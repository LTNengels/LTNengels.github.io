9.3 Data voorbereiden (met maanden-filter)

filtered_data <- covid_data %>%
  filter(country == params$land) %>%
  mutate(
    year = as.numeric(str_sub(year_week, 1, 4)),
    week = as.numeric(str_remove(year_week, ".*-")),
    
    # week → datum → maand
    date = as.Date(paste(year, week, 1, sep = "-"), "%Y-%U-%u"),
    month = as.numeric(format(date, "%m")),
    
    weekly_count = as.numeric(weekly_count)
  ) %>%
  filter(
    year == params$jaar,
    month %in% params$maanden,
    !is.na(weekly_count)
  )

cases_data <- filtered_data %>%
  filter(indicator == "cases") %>%
  arrange(week)

ggplot(cases_data, aes(x = week, y = weekly_count)) +
  geom_line(color = "blue") +
  geom_point(color = "blue") +
  labs(
    title = paste("Weekly COVID cases in", params$land),
